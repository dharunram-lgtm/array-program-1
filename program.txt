    for (int i = 0; i < r; i++) {
        for (int j = 0; j < c; j++) {
            scanf("%d", & arr[i][j]);
        }
    }
    int mi = 0, maxsum = 0;
    for (int i = 0; i < r; i++) {
        int sum = 0;
        for (int j = 0; j < c; j++) {
            sum += arr[i][j];

        }
        if (sum > maxsum)
        {
            maxsum = sum;
            mi = i;
        }
    }

}

}

int main() {
    int r,c;
    scanf("%d%d",&r,&c);
    int arr[r][c];
    for(int i=0;i<r;i++){
        for(int j=0;j<c;j++){
            scanf("%d",&arr[i][j]);
        }
    }
    for(int i=0;i<r;i++){
        for(int j=i+1;j<c;j++){
            int temp=arr[i][j];
            arr[i][j]=arr[j][i];
            arr[j][i]=temp;
        }
    }
    for(int i=0;i<r;i++){
        for(int j=0;j<c;j++){
            int temp=arr[i][0];
            arr[i][0]=arr[i][c-1];
            arr[i][c-1]=temp;
        }
    }
    for(int i=0;i<r;i++){
        for(int j=0;j<c;j++){
            printf("%d ",arr[i][j]);
        }
        printf("\n");
    }
    
}

#include <stdio.h>

int main() {
        int r, c;
        scanf("%d%d", & r, & c);
        int arr[r][c];
        for (int i = 0; i < r; i++) {
                for (int j = 0; j < c; j++) {
                        scanf("%d", & arr[i][j]);
                }
        }
        for (int i = 0; i < r; i++) {
                for (int j = i + 1; j < c; j++) {
                        int temp = arr[i][j];
                        arr[i][j] = arr[j][i];
                        arr[j][i] = temp;
                }

        }
        for (int i = 0; i < r; i++) {
                for (int j = 0; j < c; j++) {
                        int temp = arr[i][0];
                        arr[i][0] = arr[i][r - 1];
                        arr[i][r - 1] = temp;

                }

        }
        for (int i = 0; i < r; i++) {
                for (int j = 0; j < c; j++) {
                        printf("%d ", arr[i][j]);
                }

                printf("\n");
        }

}
